% B2 -- B + more air (D-like): free text indented to the content column,
%       larger gaps between sections/entries, slightly looser leading.
\documentclass[a4paper,11pt]{article}
\input{styles/common}
\usepackage[a4paper,margin=2.2cm,top=2cm]{geometry}
\setmainfont{Libertinus Serif}
\linespread{1.12}
\newlength{\datew}\setlength{\datew}{2.4cm}
\AtBeginDocument{\fontsize{10.4}{12.4}\selectfont}

\newcommand{\cvheader}[3]{%
  {\color{accent}\fontsize{26}{30}\selectfont\scshape #1}\par\vspace{5pt}
  {#2}\par\vspace{2pt}
  {\small\color{muted} #3}\par\vspace{14pt}}
\newcommand{\cvsection}[1]{%
  \par\needspace{5\baselineskip}\vspace{15pt}{\color{accent}\large\scshape #1}\par\vspace{1pt}
  {\color{rulegray}\hrule height 0.4pt}\par\vspace{9pt}}
\newcommand{\cvsubsection}[1]{\par\needspace{3\baselineskip}\vspace{4pt}\hspace*{\datew}{\itshape\color{accent} #1}\par\vspace{2pt}}
\newcommand{\cvdate}[1]{\makebox[\datew][l]{\color{muted}\small #1}}
\newcommand{\cventry}[4]{%
  \par\hangindent=\datew\hangafter=1\noindent\cvdate{#1}\textbf{#2}\hfill\textcolor{muted}{\small #3}\par
  {\leftskip=\datew\small #4\par}\vspace{7pt}}
\newcommand{\cvitem}[2]{\par\hangindent=\datew\hangafter=1\noindent\cvdate{#1}#2\par\vspace{3pt}}
\newcommand{\cvlabel}[2]{\par\hangindent=\datew\hangafter=1\noindent\makebox[\datew][l]{\small\bfseries #1}#2\par\vspace{3pt}}
\newcommand{\cvpub}[1]{\par{\leftskip=\datew\hangindent=1em\hangafter=1 #1\par}\vspace{4pt}}
\newcommand{\cvpara}[1]{\par{\leftskip=\datew #1\par}\vspace{6pt}}
